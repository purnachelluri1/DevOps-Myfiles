pipeline {
    agent {
        label 'dev'
    }
    stages {
        stage ('code') {
            steps {
                git "https://github.com/purnachelluri1/python-code-library-app.git"     
            }
        }
        stage('CQA') {
            steps {
                script {
                    def scannerHome = tool 'SonarScanner';
                    withSonarQubeEnv('mysonar') {
                        sh "${scannerHome}/bin/sonar-scanner"
                    }
                }
            }

        }
        stage('Docker Image Build') {
            steps {
                sh 'docker build -t db:v1 ./database'
		        sh 'docker build -t auth:v1 ./auth'
		        sh 'docker build -t book:v1 ./book'
		        sh 'docker build -t borrow:v1 ./borrow'
		        sh 'docker build -t frontend:v1 .'
            }
        }
        stage ('trivy scan') {
            steps {
                sh 'trivy image db:v1 >> db-report.txt'
                sh 'trivy image auth:v1 >> auth-report.txt'
                sh 'trivy image book:v1 >> book-report.txt'
                sh 'trivy image borrow:v1 >> borrow-list.txt'
                sh 'trivy image frontend:v1 >> frontend-log.txt'
            }
        }
        stage ('dockerhub') {
            steps {
                script {
                    withDockerRegistry(credentialsId: 'dockerhub') {
                        sh 'docker tag db:v1 vasuch2626/python-app-db:v1'
                        sh 'docker tag book:v1 vasuch2626/python-app-book:v1'
                        sh 'docker tag auth:v1 vasuch2626/python-app-auth:v1'
                        sh 'docker tag borrow:v1 vasuch2626/python-app-borrow:v1'
                        sh 'docker tag frontend:v1 vasuch2626/python-app-frontend:v1'
                        sh 'docker push vasuch2626/python-app-db:v1'
                        sh 'docker push vasuch2626/python-app-book:v1'
                        sh 'docker push vasuch2626/python-app-auth:v1'
                        sh 'docker push vasuch2626/python-app-borrow:v1'
                        sh 'docker push vasuch2626/python-app-frontend:v1'
                    }
                }
            }
        }
        stage ('deploy') {
            steps {
                sh 'docker stack deploy pythonapp --compose-file=compose.yml'
            }
        }
    }
}
