pipeline {
    agent {
        label 'dev'
    }
    stages {
        stage('code') {
            steps {
                git "https://github.com/purnachelluri1/python-code-library-app.git"
            }
        }
        stage('CQA') {
            steps {
                script {
                    def scannerHome = tool 'SonarScanner';
                    withSonarQubeEnv('mysonar') {
                        sh "${scannerHome}/bin/sonar-scanner"
                    }
                }
            }
        }
        stage('QualityGates') {
            steps {
                waitForQualityGate abortPipeline: true, credentialsId: 'sonar'
            }
        }
        stage('image build and push to ECR') {
            steps {
                sh 'aws ecr get-login-password --region us-east-1 | docker login --username AWS --password-stdin 762218789080.dkr.ecr.us-east-1.amazonaws.com'
                sh 'docker build -t db-v1 ./database'
                sh 'docker build -t auth-v1 ./auth'
                sh 'docker build -t book-v1 ./book'
                sh 'docker build -t borrow-v1 ./borrow'
                sh 'docker build -t frontend-v1 .'
                sh 'docker tag db-v1 762218789080.dkr.ecr.us-east-1.amazonaws.com/python-app:db-v1'
                sh 'docker tag auth-v1 762218789080.dkr.ecr.us-east-1.amazonaws.com/python-app:auth-v1'
                sh 'docker tag book-v1 762218789080.dkr.ecr.us-east-1.amazonaws.com/python-app:book-v1'
                sh 'docker tag borrow-v1 762218789080.dkr.ecr.us-east-1.amazonaws.com/python-app:borrow-v1'
                sh 'docker tag frontend-v1 762218789080.dkr.ecr.us-east-1.amazonaws.com/python-app:frontend-v1'
                sh 'docker push 762218789080.dkr.ecr.us-east-1.amazonaws.com/python-app:db-v1'
                sh 'docker push 762218789080.dkr.ecr.us-east-1.amazonaws.com/python-app:auth-v1'
                sh 'docker push 762218789080.dkr.ecr.us-east-1.amazonaws.com/python-app:book-v1'
                sh 'docker push 762218789080.dkr.ecr.us-east-1.amazonaws.com/python-app:borrow-v1'
                sh 'docker push 762218789080.dkr.ecr.us-east-1.amazonaws.com/python-app:frontend-v1'
            }
        }
        stage('image scan') {
            steps  {
                sh 'trivy image db-v1 >> db-report.txt'
                sh 'trivy image auth-v1 >> auth-report.txt'
                sh 'trivy image book-v1 >> book-report.txt'
                sh 'trivy image borrow-v1 >> borrow-list.txt'
                sh 'trivy image frontend-v1 >> frontend-log.txt'
            }
        }
        stage('deploy') {
            steps {
                sh 'docker stack deploy --with-registry-auth -c compose.yml mypythonapp'
            }
        }
    }
}
